rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    /* ── Helper functions ── */

    // যে কেউ signed in
    function isAuthenticated() {
      return request.auth != null;
    }

    // Anonymous user কিনা
    function isAnonymous() {
      return isAuthenticated() &&
        request.auth.token.firebase.sign_in_provider == 'anonymous';
    }

    // Admin কিনা — Firestore এ UID চেক করে
    function isAdmin() {
      return isAuthenticated() &&
        exists(/databases/$(database)/documents/admin_meta/config) &&
        request.auth.uid in
          get(/databases/$(database)/documents/admin_meta/config).data.adminUids;
    }

    // নিজের data কিনা
    function isOwner(uid) {
      return isAuthenticated() && request.auth.uid == uid;
    }

    // WPM valid কিনা (0–300)
    function validWPM() {
      return request.resource.data.wpm is int
          && request.resource.data.wpm >= 0
          && request.resource.data.wpm <= 300;
    }

    // Accuracy valid কিনা (0–100)
    function validAcc() {
      return request.resource.data.acc is int
          && request.resource.data.acc >= 0
          && request.resource.data.acc <= 100;
    }

    // Test duration কমপক্ষে 10 সেকেন্ড
    function validDuration() {
      return request.resource.data.duration is int
          && request.resource.data.duration >= 10;
    }


    /* ════════════════════════════════════════════
       VISITORS — কে ভিজিট করেছে
    ════════════════════════════════════════════ */
    match /visitors/{docId} {
      allow read:   if isAdmin();
      allow create: if isAuthenticated();
      allow update: if isAuthenticated()
                    && request.auth.uid == resource.data.uid;
      allow delete: if isAdmin();
    }


    /* ════════════════════════════════════════════
       SESSIONS — টাইপিং সেশন (Leaderboard ডেটা)
       Anti-cheat validation বাধ্যতামূলক
    ════════════════════════════════════════════ */
    match /sessions/{docId} {
      // যে কেউ leaderboard দেখতে পারবে
      allow read: if isAuthenticated();

      // শুধু validated data লেখা যাবে
      allow create: if isAuthenticated()
                    && validWPM()
                    && validAcc()
                    && validDuration()
                    // নিজের UID দিয়েই লিখতে হবে
                    && (
                      !('uid' in request.resource.data)
                      || request.resource.data.uid == request.auth.uid
                      || isAnonymous()
                    );

      allow update: if isAdmin();
      allow delete: if isAdmin();
    }


    /* ════════════════════════════════════════════
       DAILY STATS — দৈনিক পরিসংখ্যান
    ════════════════════════════════════════════ */
    match /daily_stats/{date} {
      allow read:   if isAuthenticated() || isAdmin();
      allow write:  if isAuthenticated();
      allow delete: if isAdmin();
    }


    /* ════════════════════════════════════════════
       FEEDBACK — যোগাযোগ ফর্ম
       যে কেউ পাঠাতে পারবে (anonymous সহ)
    ════════════════════════════════════════════ */
    match /feedback/{docId} {
      allow create: if true; // যে কেউ feedback দিতে পারবে
      allow read:   if isAdmin();
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }


    /* ════════════════════════════════════════════
       USERS — User প্রোফাইল ডেটা
    ════════════════════════════════════════════ */
    match /users/{uid} {
      // নিজের data নিজে পড়তে পারবে, admin সব পড়তে পারবে
      allow read:   if isOwner(uid) || isAdmin();

      // নিজের data নিজেই তৈরি/আপডেট করতে পারবে
      allow create: if isOwner(uid);
      allow update: if isOwner(uid);

      // delete করতে নিজে বা admin লাগবে
      allow delete: if isOwner(uid) || isAdmin();
    }


    /* ════════════════════════════════════════════
       OTP REQUESTS — OTP সংরক্ষণ (API দিয়ে তৈরি)
       Frontend directly লিখতে পারবে না
    ════════════════════════════════════════════ */
    match /otp_requests/{docId} {
      allow read:   if isAdmin();
      allow write:  if isAdmin(); // শুধু backend/admin
    }


    /* ════════════════════════════════════════════
       RATE LIMITS — Spam রোধ
    ════════════════════════════════════════════ */
    match /rate_limits/{docId} {
      allow read:   if isAdmin();
      allow write:  if isAdmin(); // শুধু backend
    }


    /* ════════════════════════════════════════════
       ADMIN META — Admin কনফিগ
       সর্বোচ্চ সুরক্ষিত — শুধু admin পড়তে/লিখতে পারবে
    ════════════════════════════════════════════ */
    match /admin_meta/{docId} {
      allow read:  if isAdmin();
      allow write: if isAdmin();
    }


    /* ════════════════════════════════════════════
       LEADERBOARD — Public read, validated write
    ════════════════════════════════════════════ */
    match /leaderboard/{docId} {
      // যে কেউ leaderboard দেখতে পারবে
      allow read: if true;

      // শুধু validated authenticated user লিখতে পারবে
      allow create: if isAuthenticated()
                    && validWPM()
                    && validAcc()
                    && validDuration()
                    && request.resource.data.uid == request.auth.uid;

      allow update: if isAdmin();
      allow delete: if isAdmin();
    }


    /* ════════════════════════════════════════════
       DENY EVERYTHING ELSE
       (যা উপরে নেই, সব deny)
    ════════════════════════════════════════════ */
    match /{document=**} {
      allow read, write: if false;
    }

  }
}
